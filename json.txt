df= spark.read.format("json").option("multiline","true").load('/Volumes/workspace/default/lpvolume/sample_nested_1000_records.json')

from pyspark.sql.functions import col, explode
df_exploded_orders = df.withColumn("order",explode(col("orders")))
df_exploded_items = df_exploded_orders.withColumn("items",explode(col("order.items")))

df_flat=df_exploded_items.select(
    col("user_id"),
    col("name"),
    col("email"),
    col("created_at"),
    col("address.street").alias("street"),
    col("address.city").alias("city"),
col("address.state").alias("state"),
col("address.zip").alias("zipcode"),
col("order.order_id").alias("order_id"),
col("order.amount").alias("order_amount"),
col("items.product_id").alias("product_id"),
col("items.quantity").alias("quantity")
)

display(df_flat)
